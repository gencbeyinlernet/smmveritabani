-- =====================================================================
-- MÜŞTERİ VE EVRAK TAKİP – SUPABASE VERİTABANI KURULUMU
-- Nasıl çalıştırılır: Supabase > SQL Editor > New query
-- Bu dosyanın TAMAMINI yapıştırın ve Run düğmesine basın.
-- Birden fazla çalıştırılırsa zarar vermez.
-- =====================================================================

create extension if not exists pgcrypto with schema extensions;

-- 1) MÜŞTERİLER
create table if not exists public.musteriler (
  id                uuid primary key default gen_random_uuid(),
  unvan             text not null check (length(trim(unvan)) > 0),
  vergi_no          text not null default '',
  sektor            text not null default '',
  mukellefiyet_turu text not null default '',
  yetkili           text not null default '',
  telefon           text not null default '' check (telefon ~ '^[0-9]*$'),
  eposta            text not null default '',
  aylik_ucret       numeric(12,2) not null default 0 check (aylik_ucret >= 0),
  beyannameler      text[] not null default array['KDV','MUHTASAR','GECICI']
                    check (beyannameler <@ array['KDV','MUHTASAR','GECICI']),
  olusturulma       timestamptz not null default now()
);

-- 2) AYLIK EVRAK TAKİBİ (her müşteri + ay için tek kayıt)
create table if not exists public.evraklar (
  musteri_id   uuid not null references public.musteriler(id) on delete cascade,
  donem        text not null check (donem ~ '^[0-9]{4}-(0[1-9]|1[0-2])$'),  -- ör. 2026-11
  geldi        boolean not null default false,
  gelis_tarihi date,
  guncelleme   timestamptz not null default now(),
  primary key (musteri_id, donem),
  check (not geldi or gelis_tarihi is not null)
);

-- 3) BEYANNAME DURUMLARI (her müşteri + tür + dönem için tek kayıt)
create table if not exists public.beyanname_kayitlari (
  musteri_id uuid not null references public.musteriler(id) on delete cascade,
  tur        text not null check (tur in ('KDV','MUHTASAR','GECICI')),
  donem      text not null,  -- KDV/Muhtasar: 2026-11, Geçici Vergi: 2026-Q3
  durum      text not null default 'Bekliyor'
             check (durum in ('Bekliyor','Hazırlandı','Onaylandı','Gönderildi')),
  guncelleme timestamptz not null default now(),
  primary key (musteri_id, tur, donem),
  check (
    (tur =  'GECICI' and donem ~ '^[0-9]{4}-Q[1-3]$') or
    (tur <> 'GECICI' and donem ~ '^[0-9]{4}-(0[1-9]|1[0-2])$')
  )
);

-- 4) KULLANICI PROFİLLERİ (kullanıcı adı, onay ve yönetici bilgisi)
create table if not exists public.profiller (
  id            uuid primary key references auth.users(id) on delete cascade,
  kullanici_adi text not null unique,
  onayli        boolean not null default false,
  yonetici      boolean not null default false,
  olusturulma   timestamptz not null default now()
);

-- 5) YARDIMCI KONTROLLER
create or replace function public.onayli_mi() returns boolean
language sql stable security definer set search_path = public as $$
  select coalesce((select onayli from public.profiller where id = auth.uid()), false);
$$;

create or replace function public.yonetici_mi() returns boolean
language sql stable security definer set search_path = public as $$
  select coalesce((select onayli and yonetici from public.profiller where id = auth.uid()), false);
$$;

-- 6) YENİ KAYIT OLAN HER KULLANICI İÇİN PROFİL OLUŞTUR
--    Yeni kullanıcılar ONAYSIZ başlar; yönetici onaylayana kadar veri göremez.
--    İstisna: henüz hiç yönetici yokken 'novarge' adıyla kayıt olan hesap
--    otomatik olarak onaylı yönetici olur.
create or replace function public.yeni_kullanici_profili() returns trigger
language plpgsql security definer set search_path = public as $$
declare
  ad   text := lower(coalesce(new.raw_user_meta_data->>'kullanici_adi', split_part(new.email, '@', 1)));
  ilk  boolean := ad = 'novarge' and not exists (select 1 from public.profiller where yonetici);
begin
  insert into public.profiller (id, kullanici_adi, onayli, yonetici)
  values (new.id, ad, ilk, ilk);
  return new;
end;
$$;

drop trigger if exists kullanici_olusunca on auth.users;
create trigger kullanici_olusunca
  after insert on auth.users
  for each row execute function public.yeni_kullanici_profili();

-- 7) SATIR DÜZEYİNDE GÜVENLİK (RLS)
--    Veri tablolarını yalnızca giriş yapmış VE onaylanmış kullanıcılar görür/değiştirir.
alter table public.musteriler          enable row level security;
alter table public.evraklar            enable row level security;
alter table public.beyanname_kayitlari enable row level security;
alter table public.profiller           enable row level security;

drop policy if exists "onayli kullanicilar" on public.musteriler;
create policy "onayli kullanicilar" on public.musteriler
  for all to authenticated using (public.onayli_mi()) with check (public.onayli_mi());

drop policy if exists "onayli kullanicilar" on public.evraklar;
create policy "onayli kullanicilar" on public.evraklar
  for all to authenticated using (public.onayli_mi()) with check (public.onayli_mi());

drop policy if exists "onayli kullanicilar" on public.beyanname_kayitlari;
create policy "onayli kullanicilar" on public.beyanname_kayitlari
  for all to authenticated using (public.onayli_mi()) with check (public.onayli_mi());

-- Herkes kendi profilini görür; yönetici herkesi görür ve değiştirir.
drop policy if exists "kendi profili veya yonetici" on public.profiller;
create policy "kendi profili veya yonetici" on public.profiller
  for select to authenticated using (id = auth.uid() or public.yonetici_mi());

drop policy if exists "yonetici gunceller" on public.profiller;
create policy "yonetici gunceller" on public.profiller
  for update to authenticated
  using (public.yonetici_mi() and id <> auth.uid())
  with check (public.yonetici_mi() and id <> auth.uid());

-- Giriş yapmamış ziyaretçilerin hiçbir tabloya erişimi yok.
revoke all on public.musteriler, public.evraklar, public.beyanname_kayitlari, public.profiller from anon;

-- 8) YÖNETİCİ İŞLEMLERİ: şifre sıfırlama ve kullanıcı silme
create or replace function public.yonetici_sifre_degistir(hedef uuid, yeni_sifre text) returns void
language plpgsql security definer set search_path = public, extensions as $$
begin
  if not public.yonetici_mi() then raise exception 'Bu işlem için yönetici yetkisi gerekir.'; end if;
  if length(yeni_sifre) < 6 then raise exception 'Şifre en az 6 karakter olmalıdır.'; end if;
  update auth.users
     set encrypted_password = extensions.crypt(yeni_sifre, extensions.gen_salt('bf')),
         updated_at = now()
   where id = hedef;
end;
$$;

create or replace function public.yonetici_kullanici_sil(hedef uuid) returns void
language plpgsql security definer set search_path = public as $$
begin
  if not public.yonetici_mi() then raise exception 'Bu işlem için yönetici yetkisi gerekir.'; end if;
  if hedef = auth.uid() then raise exception 'Kendi hesabınızı silemezsiniz.'; end if;
  delete from auth.users where id = hedef;
end;
$$;

revoke execute on function public.yonetici_sifre_degistir(uuid, text) from public, anon;
revoke execute on function public.yonetici_kullanici_sil(uuid)        from public, anon;
grant  execute on function public.yonetici_sifre_degistir(uuid, text) to authenticated;
grant  execute on function public.yonetici_kullanici_sil(uuid)        to authenticated;

-- 8b) KAYIT OLMA: kullanıcı adı + şifre ile yeni hesap açar (e-posta gerekmez).
--     Yeni hesap ONAYSIZ başlar (6. bölümdeki kural). Şifre şifrelenerek saklanır.
create or replace function public.kayit_ol(yeni_kullanici_adi text, yeni_sifre text) returns void
language plpgsql security definer set search_path = public, extensions as $$
declare
  ad      text := lower(trim(yeni_kullanici_adi));
  eposta  text;
  yeni_id uuid := gen_random_uuid();
begin
  if ad !~ '^[a-z0-9._-]{3,30}$' then
    raise exception 'Kullanıcı adı geçersiz.';
  end if;
  if length(coalesce(yeni_sifre, '')) < 6 then
    raise exception 'Şifre en az 6 karakter olmalıdır.';
  end if;
  eposta := ad || '@kullanici.invalid';
  if exists (select 1 from auth.users where email = eposta) then
    raise exception 'Bu kullanıcı adı zaten alınmış.';
  end if;

  insert into auth.users (
    instance_id, id, aud, role, email, encrypted_password, email_confirmed_at,
    raw_app_meta_data, raw_user_meta_data, created_at, updated_at,
    confirmation_token, recovery_token, email_change_token_new, email_change
  ) values (
    '00000000-0000-0000-0000-000000000000', yeni_id, 'authenticated', 'authenticated',
    eposta, extensions.crypt(yeni_sifre, extensions.gen_salt('bf')), now(),
    '{"provider":"email","providers":["email"]}'::jsonb,
    jsonb_build_object('kullanici_adi', ad), now(), now(),
    '', '', '', ''
  );

  insert into auth.identities (id, user_id, provider_id, identity_data, provider, last_sign_in_at, created_at, updated_at)
  values (
    gen_random_uuid(), yeni_id, yeni_id::text,
    jsonb_build_object('sub', yeni_id::text, 'email', eposta, 'email_verified', true),
    'email', now(), now(), now()
  );
end;
$$;

revoke execute on function public.kayit_ol(text, text) from public;
grant  execute on function public.kayit_ol(text, text) to anon, authenticated;
